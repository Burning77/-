cmake_minimum_required(VERSION 3.24...3.30)
project(InterviewAssistant VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets Multimedia Network Sql Test)
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.h)
list(FILTER CORE_SOURCES EXCLUDE REGEX "src/main\\.cpp$")
add_library(interview_core STATIC ${CORE_SOURCES})
target_include_directories(interview_core PUBLIC src)
target_compile_definitions(interview_core PUBLIC INTERVIEW_APP_VERSION="${PROJECT_VERSION}")
target_link_libraries(interview_core PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::Network Qt6::Sql)
if(WIN32)
 target_link_libraries(interview_core PUBLIC ole32 uuid avrt advapi32 winhttp bcrypt crypt32)
endif()
add_executable(InterviewAssistant WIN32 src/main.cpp)
target_link_libraries(InterviewAssistant PRIVATE interview_core)
enable_testing()
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/tst_*.cpp)
foreach(source ${TEST_SOURCES})
 get_filename_component(name ${source} NAME_WE)
 add_executable(${name} ${source})
 target_link_libraries(${name} PRIVATE interview_core Qt6::Test)
 add_test(NAME ${name} COMMAND ${name} -o ${CMAKE_CURRENT_BINARY_DIR}/${name}.txt,txt)
 set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endforeach()
